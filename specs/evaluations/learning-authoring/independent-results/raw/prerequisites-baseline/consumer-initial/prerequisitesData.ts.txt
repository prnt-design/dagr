export const PREREQUISITE_EDGES: readonly ExplorerEdge[] = PREREQUISITE_PAIRS.map(
  ([source, target]) => ({ id: `${source}->${target}`, source, target }),
);

export const PREREQUISITES_OF: ReadonlyMap<string, readonly string[]> = new Map(
  CONCEPTS.map((concept) => [
    concept.id,
    PREREQUISITE_PAIRS.filter(([, target]) => target === concept.id).map(([source]) => source),
  ]),
);

function viewOf(
  id: string,
  label: string,
  description: string,
  ids: readonly string[] | null,
  direction: 'right' | 'down',
): ExplorerView<ConceptNode> {
  const nodes = ids === null ? CONCEPTS : CONCEPTS.filter((concept) => ids.includes(concept.id));
  const present = new Set(nodes.map((node) => node.id));
  return {
    id,
    label,
    description,
    nodes,
    edges: PREREQUISITE_EDGES.filter((edge) => present.has(edge.source) && present.has(edge.target)),
    layout: { direction, edgeStyle: 'smooth' },
  };
}
