export interface SourceRef {
  readonly file: string;
  readonly field: string;
}

export type NodeKind = 'package' | 'external' | 'stage';

export interface CodebaseNode extends ExplorerNode {
  readonly kind: NodeKind;
  /** One line, shown on the node at the rich tier. */
  readonly summary: string;
  /** What is it? */
  readonly explanation: string;
  /** Why does it matter to a new contributor? */
  readonly whyItMatters: string;
  /** Where it lives in the repository, when the packet says so. */
  readonly location?: string | undefined;
  /** Marked as an illustration in the drawer; not a claim from the packet. */
  readonly example?: string | undefined;
  /** What the packet does not settle. */
  readonly uncertainty?: string | undefined;
  readonly terms: readonly string[];
  readonly sources: readonly SourceRef[];
}

export type EdgeKind = 'dependency' | 'peer' | 'optional-peer' | 'dev' | 'flow';

export interface CodebaseEdge extends ExplorerEdge {
  readonly kind: EdgeKind;
  readonly label: string;
  readonly detail: string;
  readonly inferred?: boolean | undefined;
  readonly sources: readonly SourceRef[];
}

const pj = (dir: string, field: string): SourceRef => ({
  file: `packages/${dir}/package.json`,
  field,
});

const inEvery = (field: string): readonly SourceRef[] => ALL_PACKAGES.map((dir) => pj(dir, field));
