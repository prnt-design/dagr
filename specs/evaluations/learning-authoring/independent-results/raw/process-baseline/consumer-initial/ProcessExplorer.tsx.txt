import type { CSSProperties, ReactElement } from 'react';
import { DagrExplorer } from '@prnt/dagr-explorer';
import type { ExplorerDetailsContext } from '@prnt/dagr-explorer';
import '@prnt/dagr-explorer/styles.css';

import {
  KIND_COLOR,
  KIND_LABEL,
  PROCESS_VIEWS,
  SOURCE_SNAPSHOT,
  type ProcessEdge,
  type ProcessNode,
} from './process-data';

const searchText = (node: ProcessNode): string =>
  [node.id, node.label, KIND_LABEL[node.kind], node.summary, node.command ?? '', ...(node.detail ?? [])].join(
    ' ',
  );

const firstNodeOf = (view: { readonly nodes: readonly ProcessNode[] }): string | null =>
  view.nodes[0]?.id ?? null;

const styles = {
  page: {
    fontFamily: 'system-ui, -apple-system, "Segoe UI", sans-serif',
    color: '#111827',
    maxWidth: 1200,
    margin: '0 auto',
    padding: '16px',
  },
  intro: { margin: '4px 0 12px', lineHeight: 1.5, color: '#374151' },
  legend: {
    display: 'flex',
    flexWrap: 'wrap',
    gap: 12,
    margin: '0 0 12px',
    padding: 0,
    listStyle: 'none',
    fontSize: 13,
    color: '#374151',
  },
  swatch: {
    display: 'inline-block',
    width: 10,
    height: 10,
    borderRadius: 2,
    marginRight: 6,
    verticalAlign: 'middle',
  },
  badge: {
    display: 'inline-block',
    fontSize: 11,
    fontWeight: 600,
    textTransform: 'uppercase',
    letterSpacing: '0.04em',
    color: '#fff',
    borderRadius: 4,
    padding: '1px 6px',
  },
  nodeBody: {
    display: 'flex',
    flexDirection: 'column',
    gap: 4,
    height: '100%',
    padding: '8px 10px',
    boxSizing: 'border-box',
    textAlign: 'left',
    overflow: 'hidden',
  },
  nodeTitle: { fontWeight: 600, fontSize: 14, lineHeight: 1.25 },
  nodeSummary: { fontSize: 12, lineHeight: 1.35, color: '#4b5563' },
  code: {
    display: 'block',
    fontFamily: 'ui-monospace, SFMono-Regular, Menlo, monospace',
    fontSize: 12,
    background: '#f3f4f6',
    border: '1px solid #e5e7eb',
    borderRadius: 6,
    padding: '8px 10px',
    whiteSpace: 'pre-wrap',
    wordBreak: 'break-word',
    margin: '8px 0',
  },
  quote: {
    margin: '12px 0 4px',
    padding: '6px 10px',
    borderLeft: '3px solid #d1d5db',
    color: '#4b5563',
    fontSize: 13,
  },
  sourceLine: { fontSize: 12, color: '#6b7280', margin: '0 0 12px' },
  connList: { listStyle: 'none', padding: 0, margin: '4px 0 0', display: 'grid', gap: 6 },
  connButton: {
    width: '100%',
    minHeight: 44,
    textAlign: 'left',
    padding: '6px 10px',
    borderRadius: 6,
    border: '1px solid #d1d5db',
    background: '#fff',
    cursor: 'pointer',
    font: 'inherit',
    fontSize: 13,
  },
  footer: { fontSize: 12, color: '#6b7280', marginTop: 12 },
} satisfies Record<string, CSSProperties>;

function KindBadge({ kind }: { readonly kind: ProcessNode['kind'] }): ReactElement {
  return <span style={{ ...styles.badge, background: KIND_COLOR[kind] }}>{KIND_LABEL[kind]}</span>;
}

function renderNode(node: ProcessNode, context: { readonly tier: 'summary' | 'rich' }): ReactElement {
  if (context.tier === 'summary') {
    return (
      <div style={styles.nodeBody}>
        <span style={styles.nodeTitle}>{node.label}</span>
      </div>
    );
  }
  return (
    <div style={{ ...styles.nodeBody, borderTop: `4px solid ${KIND_COLOR[node.kind]}` }}>
      <KindBadge kind={node.kind} />
      <span style={styles.nodeTitle}>{node.label}</span>
      <span style={styles.nodeSummary}>{node.summary}</span>
    </div>
  );
}

function renderDetails({
  node,
  connections,
  follow,
}: ExplorerDetailsContext<ProcessNode, ProcessEdge>): ReactElement {
  const outgoing = connections.filter((c) => c.edge.source === node.id && c.edge.target !== node.id);
  const incoming = connections.filter((c) => c.edge.target === node.id && c.edge.source !== node.id);

  const list = (heading: string, items: typeof connections, arrow: string) =>
    items.length === 0 ? null : (
      <section>
        <h3 style={{ fontSize: 13, margin: '12px 0 4px' }}>{heading}</h3>
        <ul style={styles.connList}>
          {items.map(({ edge, node: other }) => (
            <li key={edge.id}>
              <button type="button" style={styles.connButton} onClick={() => follow(other.id)}>
                <span aria-hidden="true">{arrow} </span>
                {other.label}
                {edge.label ? <span style={{ color: '#6b7280' }}> ({edge.label})</span> : null}
              </button>
            </li>
          ))}
        </ul>
      </section>
    );

  return (
    <article>
      <KindBadge kind={node.kind} />
      <h2 style={{ fontSize: 18, margin: '6px 0 8px' }}>{node.label}</h2>
      <p style={{ margin: '0 0 8px', lineHeight: 1.5 }}>{node.summary}</p>
      {node.command ? <code style={styles.code}>{node.command}</code> : null}
      {node.detail && node.detail.length > 0 ? (
        <ul style={{ margin: '8px 0', paddingLeft: 18, lineHeight: 1.5 }}>
          {node.detail.map((line) => (
            <li key={line}>{line}</li>
          ))}
        </ul>
      ) : null}
      <blockquote style={styles.quote}>{node.quote}</blockquote>
      <p style={styles.sourceLine}>Source: {node.source}</p>
      {list('Before this', incoming, '←')}
      {list('Next', outgoing, '→')}
    </article>
  );
}

export interface ProcessExplorerProps {
  readonly className?: string | undefined;
  readonly style?: CSSProperties | undefined;
}

/**
 * A walkthrough of how a change reaches main, built from the repository's
 * AGENTS.md. Each view is one angle on the same process.
 */
export function ProcessExplorer({ className, style }: ProcessExplorerProps): ReactElement {
  const kinds = Object.keys(KIND_LABEL) as ProcessNode['kind'][];
  return (
    <div className={className} style={{ ...styles.page, ...style }}>
      <h1 style={{ fontSize: 24, margin: '0 0 4px' }}>How work reaches main</h1>
      <p style={styles.intro}>
        Every change reaches <code>main</code> through a pull request; nobody pushes to <code>main</code>{' '}
        directly. Start with <strong>Path to main</strong> for the six steps, then open{' '}
        <strong>Local gate</strong>, <strong>Reviews</strong> and <strong>Guardrails</strong> for the detail.
        Select a node, or search, to read what it means and why, with the source excerpt it comes from.
      </p>
      <ul style={styles.legend} aria-label="Node kinds">
        {kinds.map((kind) => (
          <li key={kind}>
            <span aria-hidden="true" style={{ ...styles.swatch, background: KIND_COLOR[kind] }} />
            {KIND_LABEL[kind]}
          </li>
        ))}
      </ul>
      <DagrExplorer
        label="How work reaches main"
        views={PROCESS_VIEWS}
        defaultSelectedId={firstNodeOf(PROCESS_VIEWS[0]!)}
        selectOnViewChange={firstNodeOf}
        searchText={searchText}
        renderNode={renderNode}
        renderDetails={renderDetails}
        nodeAriaLabel={(node) => `${KIND_LABEL[node.kind]}: ${node.label}`}
        style={{ ['--dagr-explorer-height' as string]: '560px' } as CSSProperties}
      />
      <p style={styles.footer}>
        Source: AGENTS.md (sections “Commands”, “What agents may do unattended” and “How work reaches main”) at
        repository snapshot <code>{SOURCE_SNAPSHOT.slice(0, 12)}</code>. Dashed edges are conditional or optional
        paths.
      </p>
    </div>
  );
}

export default ProcessExplorer;
