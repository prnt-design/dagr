import type { ExplorerEdge, ExplorerNode, ExplorerView } from '@prnt/dagr-explorer';

export type StepKind = 'step' | 'command' | 'decision' | 'outcome' | 'review' | 'rule';

export interface ProcessNode extends ExplorerNode {
  readonly kind: StepKind;
  /** One sentence shown on the node and at the top of the drawer. */
  readonly summary: string;
  /** Longer explanation, paraphrasing the source. */
  readonly detail?: readonly string[] | undefined;
  /** A command to run, verbatim from the source. */
  readonly command?: string | undefined;
  /** Where in the source packet the claim comes from. */
  readonly source: string;
  /** A verbatim excerpt from that source. */
  readonly quote: string;
}

export type ProcessEdge = ExplorerEdge;
