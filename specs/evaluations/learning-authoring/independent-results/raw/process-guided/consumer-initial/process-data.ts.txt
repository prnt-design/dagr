import type { ExplorerEdge, ExplorerNode, ExplorerView } from '@prnt/dagr-explorer';

export type StepKind = 'step' | 'check' | 'review' | 'outcome' | 'rule';

export interface SourceRef {
  /** Section of AGENTS.md at snapshot ccb66cad9dfc3baf650d65338e2bb2ae6c38aae2. */
  readonly section: string;
  /** Verbatim excerpt (line breaks collapsed to spaces). */
  readonly quote: string;
}

export interface ProcessNode extends ExplorerNode {
  readonly kind: StepKind;
  /** One line, shown on the node at the rich tier. */
  readonly summary: string;
  readonly explanation: string;
  readonly whyItMatters: string;
  readonly command?: string | undefined;
  readonly sources: readonly SourceRef[];
  readonly terms?: readonly string[] | undefined;
}

export interface ProcessEdge extends ExplorerEdge {
  readonly label: string;
}

  step: '#2563eb',
  check: '#0f766e',
  review: '#7c3aed',

export const KIND_LABEL: Record<StepKind, string> = {
  step: 'Step',
  check: 'Check',
  review: 'Review',
  outcome: 'Outcome',
  rule: 'Rule',

function node(n: Omit<ProcessNode, 'color'>): ProcessNode {
  return { ...n, color: KIND_COLOR[n.kind] };
}
