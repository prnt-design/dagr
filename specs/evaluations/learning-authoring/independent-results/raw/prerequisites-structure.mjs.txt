import ts from '/tmp/dagr-learning-consumer/node_modules/typescript/lib/typescript.js';
import {readFileSync,writeFileSync} from 'node:fs';
const root='/tmp/dagr-learning-independent-collected';const result=[];
for(const arm of ['baseline','guided']){
 const dir=`${root}/prerequisites-${arm}`;
 const name=arm==='baseline'?'prerequisitesData.ts':'prerequisites-data.ts';
 const code=ts.transpileModule(readFileSync(`${dir}/initial/${name}`,'utf8'),{compilerOptions:{module:ts.ModuleKind.ESNext,target:ts.ScriptTarget.ES2022}}).outputText;
 writeFileSync(`${dir}/data-validation.mjs`,code);const data=await import(`${dir}/data-validation.mjs`);
 const views=data.CONCEPT_VIEWS??data.views;
 result.push({arm,views:views.map(v=>({id:v.id,nodes:v.nodes.length,edges:v.edges.length,uniqueNodes:new Set(v.nodes.map(n=>n.id)).size===v.nodes.length,uniqueEdges:new Set(v.edges.map(e=>e.id)).size===v.edges.length,endpoints:v.edges.every(e=>v.nodes.some(n=>n.id===e.source)&&v.nodes.some(n=>n.id===e.target))})),domainSearch:views[0].nodes.filter(n=>(arm==='baseline'?`${n.label} ${n.stage} ${n.summary}`:[n.id,n.label,n.kind,n.summary,n.explanation,...n.keywords].join(' ')).toLowerCase().includes('missing_edge_endpoint')).map(n=>n.id)});
}
writeFileSync(`${root}/prerequisites-structure-results.json`,JSON.stringify(result,null,2));
