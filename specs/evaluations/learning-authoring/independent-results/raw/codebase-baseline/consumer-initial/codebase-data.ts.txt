export type NodeKind = 'package' | 'workspace' | 'external' | 'entry';

export interface Fact {
  readonly label: string;
  readonly value: string;
}

export interface CodebaseNode extends ExplorerNode {
  readonly kind: NodeKind;
  readonly summary: string;
  readonly facts: readonly Fact[];
  /** The packet file(s) every fact on this node comes from. */
  readonly manifest: readonly string[];
}

export type Relation = 'dependency' | 'peer' | 'optional-peer' | 'dev' | 'export';

export interface CodebaseEdge extends ExplorerEdge {
  readonly relation: Relation;
  readonly range: string;
  readonly manifest: string;
}

function packageNode(
  key: PackageKey,
  name: string,
  description: string,
  summary: string,
  extra: readonly Fact[],
  test = 'vitest run',
): CodebaseNode {
  return {
    id: key,
    label: name,
    kind: 'package',
    color: KIND_COLOR.package,
    summary,
    manifest: [M[key]],
    facts: [
      { label: 'description', value: description },
      { label: 'version', value: '0.2.0' },
      { label: 'license', value: 'MIT' },
      { label: 'type', value: 'module' },
      { label: 'directory', value: `packages/${key}` },
      ...extra,
      { label: 'script: test', value: test },
      ...COMMON_SCRIPTS.map((s) => ({ label: `script: ${s.label}`, value: s.value })),
    ],
  };
}

function external(
  id: string,
  label: string,
  summary: string,
  manifest: readonly string[],
  facts: readonly Fact[] = [],
): CodebaseNode {
  return {
    id,
    label,
    kind: 'external',
    color: KIND_COLOR.external,
    summary,
    manifest,
    facts,
    size: { width: 200, height: 80 },
  };
}

function edge(
  from: string,
  to: string,
  relation: Relation,
  range: string,
  manifest: string,
): CodebaseEdge {
  const dashed = relation === 'dev' || relation === 'optional-peer';
  return {
    id: `${from}->${to}:${relation}`,
    source: from,
    target: to,
    relation,
    range,
    manifest,
    label: range,
    color: RELATION_COLOR[relation],
    dash: dashed,
  };
}

function entry(
  pkg: PackageKey,
  subpath: string,
  facts: readonly Fact[],
  summary: string,
): CodebaseNode {
  return {
    id: `${pkg}:${subpath}`,
    label: `${PACKAGES[pkg].label}${subpath === '.' ? '' : subpath.slice(1)}`,
    kind: 'entry',
    color: KIND_COLOR.entry,
    summary,
    manifest: [M[pkg]],
    facts: [{ label: 'export key', value: `"${subpath}"` }, ...facts],
    size: { width: 240, height: 88 },
  };
}

const packagingView: ExplorerView<CodebaseNode, CodebaseEdge> = {
  id: 'packaging',
  label: 'Published entry points',
  description:
    'The subpaths each package.json exports. Every package also sets main, module and types to ./dist/index.*.',
  layout: { direction: 'down', nodeSize: { width: 240, height: 104 }, nodeSep: 32, rankSep: 80 },
  nodes: packagingNodes,
  edges: packagingNodes
    .filter((n) => n.kind === 'entry')
    .map((n) => {
      const [pkg, subpath] = n.id.split(':') as [PackageKey, string];
      return edge(pkg, n.id, 'export', subpath, M[pkg]);
    }),
  groups: (Object.keys(PACKAGES) as PackageKey[]).map(
    (pkg): ExplorerGroup => ({
      id: `group:${pkg}`,
      label: `packages/${pkg}`,
      nodeIds: packagingNodes.filter((n) => n.id === pkg || n.id.startsWith(`${pkg}:`)).map((n) => n.id),
    }),
  ),
};
