import type { ExplorerEdge, ExplorerNode, ExplorerView } from '@prnt/dagr-explorer';

export type ConceptKind = 'Setup' | 'Data model' | 'Explorer API' | 'Reading experience';

export interface Concept extends ExplorerNode {
  readonly kind: ConceptKind;
  readonly summary: string;
  readonly explanation: string;
  readonly whyItMatters: string;
  readonly example?: { readonly code: string; readonly note: string } | undefined;
  readonly keywords: readonly string[];
  readonly sources: readonly string[];
}

/** Edges point from a prerequisite to the concept it is needed for. */
export interface Prerequisite extends ExplorerEdge {
  readonly label: 'needed for';
  readonly reason: string;
}

const byId = new Map(concepts.map((c) => [c.id, c]));

function view(id: string, label: string, description: string, ids: readonly string[]): ExplorerView<Concept, Prerequisite> {
  const members = new Set(ids);
  return {
    id,
    label,
    description,
    nodes: ids.map((nodeId) => {
      const node = byId.get(nodeId);
      if (!node) throw new Error(`Unknown concept id in view ${id}: ${nodeId}`);
      return node;
    }),
    edges: prerequisites.filter((e) => members.has(e.source) && members.has(e.target)),
    layout: { direction: 'right', nodeSize: { width: 260, height: 140 } },
  };
}
